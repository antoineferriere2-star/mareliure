import puppeteer from '../mareliure-brief6-20261020/tools/node_modules/puppeteer-core/lib/puppeteer/puppeteer-core.js';
import fs from 'node:fs';
import assert from 'node:assert/strict';
const fine=process.argv[2]==='fine';
const origin='http://127.0.0.1:'+(fine?'8092':'8091');
const dir='D:/CodexProjects/mareliure-review163-164-20261021/docs/review-163-164-20261021/';
const browser=await puppeteer.launch({executablePath:'D:/CodexData/ms-playwright/chromium-1234/chrome-win64/chrome.exe',headless:true,userDataDir:'D:/CodexProjects/outputs/mareliure-brief9-20261021/profile-review-'+(fine?'fine':'ma'),args:['--no-sandbox','--disable-background-networking','--disable-extensions']});
const receipt={at:new Date().toISOString(),source:'ef45d12c56ad322187e9cda7771317a47810eb99',brand:fine?'FINE_BINDERY':'MA_RELIURE',origin,viewports:[],blocked:[],sent:[],console:[],exceptions:[]};
try{
 for(const width of [1280,375]){
  const context=await browser.createBrowserContext();
  const page=await context.newPage();
  await page.setViewport({width,height:width===375?812:900,deviceScaleFactor:1});
  await page.setRequestInterception(true);
  page.on('request',request=>{
    const url=new URL(request.url());
    if((url.origin!==origin&&!['data:','blob:'].includes(url.protocol))||!['GET','HEAD'].includes(request.method())||url.pathname==='/api/public/track-view'){
      receipt.blocked.push({method:request.method(),url:url.origin+url.pathname});return request.abort();
    }
    if(url.pathname.startsWith('/_serverFn/')){
      const name=JSON.parse(Buffer.from(url.pathname.split('/').pop(),'base64url').toString()).export;
      if(!/^getRequest(?:MarketplaceBrand|FineBinderyLocale)_createServerFn_handler$/.test(name)){
        receipt.blocked.push({method:request.method(),name});return request.abort();
      }
    }
    receipt.sent.push({method:request.method(),url:url.origin+url.pathname});request.continue();
  });
  page.on('console',msg=>{if(['error','warn'].includes(msg.type()))receipt.console.push({width,text:msg.text()});});
  page.on('pageerror',error=>receipt.exceptions.push({width,error:error.message}));
  const rows=[];
  const snapshot=()=>page.evaluate(()=>({title:document.title,url:location.pathname+location.search,h1:document.querySelector('h1')?.textContent,tabs:[...document.querySelectorAll('[role="tab"]')].map(e=>({text:e.textContent,selected:e.getAttribute('aria-selected')})),header:document.querySelector('header')?.innerText,headerLinks:[...document.querySelectorAll('header a')].map(e=>({text:e.textContent?.trim(),href:e.getAttribute('href'),visible:e.getBoundingClientRect().width>0&&e.getBoundingClientRect().height>0})),scrollWidth:document.documentElement.scrollWidth,innerWidth:innerWidth}));
  for(const locale of fine?['en','fr','de','it','es']:['fr']){
    await page.goto(origin+'/auth?space=atelier'+(fine?'&locale='+locale:''),{waitUntil:'networkidle0',timeout:180000});
    await page.waitForSelector('[role="tab"][aria-selected="true"]',{timeout:30000});
    const initial=await snapshot();
    assert.equal(initial.tabs[1].selected,'true');
    assert.match(initial.title,fine?/Workshop space/:/Mon espace atelier/);
    if(locale==='en'||!fine)await page.screenshot({path:dir+`auth-${fine?'fine':'ma'}-${width}-atelier.png`});
    const tabs=await page.$$('[role="tab"]');await tabs[0].click();
    await page.waitForFunction(()=>document.querySelectorAll('[role="tab"]')[0]?.getAttribute('aria-selected')==='true');
    const customer=await snapshot();
    if(locale==='en'||!fine)await page.screenshot({path:dir+`auth-${fine?'fine':'ma'}-${width}-client.png`});
    await tabs[1].click();
    const back=await snapshot();
    rows.push({locale,initial,customer,back});
  }
  if(!fine){
    await page.goto(origin+'/auth',{waitUntil:'networkidle0',timeout:90000});await page.waitForSelector('[role="tab"]');
    const customerFirst=await snapshot();await (await page.$$('[role="tab"]'))[1].click();
    rows.push({customerFirst,switchedAtelier:await snapshot()});
    await page.goto(origin+'/partenaires-relieurs',{waitUntil:'networkidle0',timeout:180000});
    await page.waitForFunction(async()=>{const path='/src/lib/hydration.ts';return(await import(path)).isHydrated();});
    const anchor=await page.$('a[href="#candidature"]');assert(anchor);
    const before=await anchor.evaluate(el=>({text:el.textContent,top:el.getBoundingClientRect().top,left:el.getBoundingClientRect().left,width:el.getBoundingClientRect().width}));
    await page.screenshot({path:dir+`partners-${width}-hero.png`});
    await anchor.click();
    await page.waitForFunction(()=>location.hash==='#candidature');
    await page.waitForFunction(()=>{const r=document.getElementById('candidature').getBoundingClientRect();return r.top<innerHeight&&r.bottom>0;});
    rows.push({candidature:{before,after:await page.evaluate(()=>({hash:location.hash,top:document.getElementById('candidature').getBoundingClientRect().top,heading:document.getElementById('candidature').querySelector('h2')?.textContent,scrollWidth:document.documentElement.scrollWidth,innerWidth:innerWidth}))}});
    await page.screenshot({path:dir+`partners-${width}-candidature.png`});
  }
  receipt.viewports.push({width,rows});await context.close();
 }
}finally{
 receipt.finishedAt=new Date().toISOString();
 fs.writeFileSync(dir+(fine?'fine':'ma')+'-ui.json',JSON.stringify(receipt,null,2)+'\n');
 await browser.close();
}
console.log(JSON.stringify({brand:receipt.brand,viewports:receipt.viewports.map(v=>({width:v.width,rows:v.rows.length})),exceptions:receipt.exceptions,blocked:receipt.blocked.length}));
